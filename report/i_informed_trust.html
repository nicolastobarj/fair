Estrategia 2. Tricotomizar y modelo multinomial

library(nnet)

data <- data %>%
  mutate(
    z_know = as.numeric(scale(fonasa_comp_objetiva_pos)),
    z_trust_func = as.numeric(scale(fonasa_conf_belief_funcional_pos))
  )

data <- data %>%
  mutate(
    calibrated_misinfo_func = case_when(
      z_know >= 0 & z_trust_func >= 0 ~ "Informed trust",
      z_know <  0 & z_trust_func >= 0 ~ "Ignorant trust",
      z_trust_func < 0              ~ "Skeptical informed / mistrust uninformed"
    )
  )

data <- data %>%
  mutate(
    z_trust_moral = as.numeric(scale(fonasa_conf_belief_moral_pos)),
    calibrated_misinfo_moral = case_when(
      z_know >= 0 & z_trust_moral >= 0 ~ "Informed trust",
      z_know <  0 & z_trust_moral >= 0 ~ "Ignorant trust",
      z_trust_moral < 0               ~ "Skeptical informed / mistrust uninformed"
    )
  )

data <- data %>%
  mutate(
    calibrated_funcional_cat = relevel(
      factor(calibrated_misinfo_moral),
      ref = "Skeptical informed / mistrust uninformed"
    ),
    calibrated_moral_cat = relevel(
      factor(calibrated_misinfo_moral),
      ref = "Skeptical informed / mistrust uninformed"
    )
  )

ivs_intervencion <- c(
  "ai_disclosure_fonasa",
  "control_humano_fonasa",
  "performance_fonasa"
)

ivs_demograficas <- c(
  "genero",
  "edad_num",
  "education_recoded"
)
