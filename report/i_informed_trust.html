library(nnet)

data <- data %>%
  mutate(
    z_know_fonasa = as.numeric(scale(fonasa_comp_objetiva_pos)),
    z_trust_func_fonasa = as.numeric(scale(fonasa_conf_belief_funcional_pos)),
    z_trust_moral_fonasa = as.numeric(scale(fonasa_conf_belief_moral_pos))
  )

data <- data %>%
  mutate(
    fonasa_calibrated_misinfo_func = case_when(
      z_know_fonasa >= 0 & z_trust_func_fonasa >= 0 ~ "Informed trust",
      z_know_fonasa < 0 & z_trust_func_fonasa >= 0 ~ "Ignorant trust",
      z_trust_func_fonasa < 0 ~ "Skeptical informed / mistrust uninformed"
    )
  )

data <- data %>%
  mutate(
    fonasa_calibrated_misinfo_moral = case_when(
      z_know_fonasa >= 0 & z_trust_moral_fonasa >= 0 ~ "Informed trust",

      z_trust_moral_fonasa < 0 ~ "Skeptical informed / mistrust uninformed"
    )
  )
